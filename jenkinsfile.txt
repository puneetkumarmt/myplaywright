pipeline {

    stages {
        stage('Checkout') {
            steps {
                // Pulls code from your configured source control repository
                checkout scm
            }
        }
        stage('Run Playwright Tests') {
            steps {
                // Executes your test suite. 
                // Note: The Docker image already has the browsers pre-installed.
                sh 'npx playwright test'
            }
        }
    }

    post {
        always {
            // Archives test reports and screenshots so they are accessible directly in the Jenkins UI
            archiveArtifacts artifacts: 'playwright-report/**, test-results/**', allowEmptyArchive: true
        }
    }
}
